Gráfico de barras de porcentaje de germinación

library(dplyr)
library(ggplot2)

gsm %>%
  group_by(Escarificacion) %>%
  summarise(
    mean_grp = mean(grp, na.rm = TRUE),
    sd_grp = sd(grp, na.rm = TRUE),
    n = n()
  ) %>%
  mutate(se_grp = sd_grp / sqrt(n)) %>%
  ggplot(aes(x = Escarificacion, y = mean_grp, fill = Escarificacion)) +
  geom_col(position = "dodge", width = 0.5) +
  geom_errorbar(aes(ymin = mean_grp - se_grp, ymax = mean_grp + se_grp), width = 0.2, color = "black") +
  scale_fill_brewer(palette = "Set2") +
  labs(title = ("Germinación (%grp) en semillas de Tamarindo"),
       y = "Porcentaje de Germinación (grp)",
       x = "Tratamiento",
       fill = "Tratamiento") +
  theme_bw()
